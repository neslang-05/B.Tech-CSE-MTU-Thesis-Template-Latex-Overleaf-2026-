% APPENDIX B: REST API SPECIFICATION
% =========================================================================
This appendix provides the REST API contract specifications for the system's core services. The architecture decouples the frontend client from the backend and machine learning services using standardized HTTP request and response structures.

\section*{1. User Authentication Endpoint}
\subsection*{POST /api/v1/auth/login}
Authenticates users and returns a JSON Web Token (JWT) for secure session management.

\begin{table}[htbp]
  \centering
  \caption{Specification for \texttt{POST /api/v1/auth/login}}
  \label{tab:api-login}
  \small
  \renewcommand{\arraystretch}{1.1}
  \begin{tabularx}{\textwidth}{|l|l|X|}
    \hline
    \textbf{Parameter} & \textbf{Type / Format} & \textbf{Description} \\
    \hline
    \textbf{Headers} & \texttt{application/json} & Request payload must be formatted as JSON. \\
    \textbf{Payload: username} & String (Required) & Username or email address of the registered user. \\
    \textbf{Payload: password} & String (Required) & Case-sensitive raw password. \\
    \hline
    \textbf{Success Response} & \texttt{200 OK} & Returns: \texttt{\{ "token": "JWT\_STRING", "expires": 3600 \}} \\
    \textbf{Error Response} & \texttt{401 Unauthorized} & Triggered by invalid credentials or inactive accounts. \\
    \hline
  \end{tabularx}
\end{table}

\section*{2. Machine Learning Inference Endpoint}
\subsection*{POST /api/v1/predict}
Executes real-time model inference on user-uploaded imagery or sensor telemetry.

\begin{table}[htbp]
  \centering
  \caption{Specification for \texttt{POST /api/v1/predict}}
  \label{tab:api-predict}
  \small
  \renewcommand{\arraystretch}{1.1}
  \begin{tabularx}{\textwidth}{|l|l|X|}
    \hline
    \textbf{Parameter} & \textbf{Type / Format} & \textbf{Description} \\
    \hline
    \textbf{Content-Type} & \texttt{multipart/form-data} & Requires binary file upload stream. \\
    \textbf{Payload: file} & Binary File (Required) & Image or data file (e.g., JPEG, PNG, CSV). \\
    \textbf{Payload: threshold} & Float (Optional) & Confidence threshold. Defaults to \texttt{0.50}. \\
    \hline
    \textbf{Success Response} & \texttt{200 OK} & Returns JSON listing prediction classes, confidence scores, and processing latency. \\
    \textbf{Error Response} & \texttt{422 Unprocessable} & Raised if the input file format is unsupported or corrupted. \\
    \hline
  \end{tabularx}
\end{table}
